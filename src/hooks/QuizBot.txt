            level: null, // primary | secondary

            class: null, // for secondary
            questions: [],

            selectedAnswers: [], // array of selected option indices

            totalQuestions: 0,

        //  MODE TOGGLE
        // ═══════════════════════════════════════════════════════════════════

        function setMode(mode) {
            state.mode = mode;
            document.querySelectorAll('.toggle-btn').forEach(btn => {
                btn.classList.toggle('active', btn.dataset.mode === mode);
            });
            updateModeLabel();
        }

        function updateModeLabel() {
            const label = $('modeLabel');
            if (state.mode === 'quiz') {
                label.textContent = '🔍 Quiz mode — test your knowledge';
            } else {
                label.textContent = '📖 Notes mode — read and revise';
            }

        let modalLevel = null;

        function openSelection(level) {
            modalLevel = level;
            const title = $('modalTitle');
            const sub = $('modalSub');
            const classGroup = $('classGroup');
            const subjectSelect = $('subjectSelect');

            if (level === 'primary') {
                title.textContent = '📚 Primary Seven';
                sub.textContent = 'Choose your term and subject.';
                classGroup.style.display = 'none';
                // populate subjects
                subjectSelect.innerHTML = PRIMARY_SUBJECTS.map(s =>
                    `<option value="${s}">${s}</option>`
                ).join('');
            } else {
                title.textContent = '🎓 Secondary';
                sub.textContent = 'Choose term, class &amp; subject.';
                classGroup.style.display = 'block';
                // populate class
                const classSelect = $('classSelect');
                const classes = ['Senior 1', 'Senior 2', 'Senior 3', 'Senior 4', 'Senior 5', 'Senior 6'];
                classSelect.innerHTML = classes.map(c =>
                    `<option value="${c}">${c}</option>`
                ).join('');
                // populate subjects for the first class
                updateSecondarySubjects('Senior 1');
                // listen to class change
                classSelect.onchange = function() {
                    updateSecondarySubjects(this.value);
                };
            }

            // reset term
            $('termSelect').value = 'Term 1';
            modal.classList.add('open');
        }

        function updateSecondarySubjects(className) {
            const subjects = SECONDARY_SUBJECTS[className] || ['English', 'Mathematics'];
            const subjectSelect = $('subjectSelect');
            subjectSelect.innerHTML = subjects.map(s =>
                `<option value="${s}">${s}</option>`
            ).join('');
        }

        function closeModal() {
            modal.classList.remove('open');
        }

        function startFromModal() {
            const term = $('termSelect').value;
            const subject = $('subjectSelect').value;
            let className = null;
            if (modalLevel === 'secondary') {
                className = $('classSelect').value;
            }

            if (!subject) {
                alert('Please select a subject.');
                return;
            }

            state.level = modalLevel;
            state.term = term;
            state.subject = subject;
            state.class = className;

            closeModal();

            if (state.mode === 'quiz') {
                startQuiz();
            } else {
                showNotesView();
            }
        }

        // close modal on overlay click
        modal.addEventListener('click', function(e) {
            if (e.target === this) closeModal();
        });

        // ═══════════════════════════════════════════════════════════════════
        //  QUIZ ENGINE
        // ═══════════════════════════════════════════════════════════════════

        function startQuiz() {
            const levelLabel = state.level === 'primary' ? 'Primary' : state.class || 'Secondary';
            const subjectLabel = state.subject;
            const termLabel = state.term;

            // Get questions
            const qs = getQuestions(state.level, state.subject, state.term);
            if (!qs || qs.length === 0) {
                alert('No questions available for this selection. Please try another subject.');
                return;
            }

            state.questions = qs;
            state.currentIndex = 0;
            state.selectedAnswers = new Array(qs.length).fill(null);
            state.score = 0;
            state.totalQuestions = qs.length;

            // Update UI
            $('quizTitle').textContent = `📝 ${subjectLabel} Quiz`;
            $('quizSubjectTag').textContent = `${levelLabel} · ${termLabel}`;

            showView('quiz');
            renderQuestion();
        }

        function renderQuestion() {
            const idx = state.currentIndex;
            const qs = state.questions;
            if (!qs || qs.length === 0) return;

            const q = qs[idx];
            const total = qs.length;

            // progress
            $('progressFill').style.width = ((idx + 1) / total * 100) + '%';
            $('questionCounter').textContent = `${idx+1}/${total}`;

            // question text
            $('questionText').textContent = q.q;

            // options
            const container = $('optionsContainer');
            const letters = ['A', 'B', 'C', 'D'];
            const selected = state.selectedAnswers[idx];

            container.innerHTML = q.opts.map((opt, i) => {
                let cls = 'qopt';
                if (selected !== null) {
                    cls += ' disabled';
                    if (i === q.ans) cls += ' correct';
                    if (i === selected && i !== q.ans) cls += ' wrong';
                    if (i === selected) cls += ' selected';
                }
                return `
              <div class="${cls}" data-index="${i}" onclick="selectAnswer(${i})">
                <span class="letter">${letters[i]}</span>
                <span class="text">${opt}</span>
              </div>
            `;
            }).join('');

            // update buttons
            const prevBtn = $('prevBtn');
            const nextBtn = $('nextBtn');
            prevBtn.style.display = idx === 0 ? 'none' : 'inline-flex';
            nextBtn.textContent = idx === total - 1 ? '📊 See Results' : 'Next →';
            nextBtn.disabled = selected === null;

            // if all questions answered, show results
            if (state.selectedAnswers.every(a => a !== null) && idx === total - 1) {
                // already answered all
            }
        }

        function selectAnswer(index) {
            const idx = state.currentIndex;
            if (state.selectedAnswers[idx] !== null) return; // already answered

            state.selectedAnswers[idx] = index;
            renderQuestion();

            // auto advance after a short delay if not last
            if (idx < state.questions.length - 1) {
                setTimeout(() => {
                    if (state.view === 'quiz') nextQuestion();
                }, 600);
            } else {
                // enable results button
                $('nextBtn').disabled = false;
            }
        }

        function nextQuestion() {
            const idx = state.currentIndex;
            const total = state.questions.length;

            // if last question, show results
            if (idx === total - 1) {
                // check if all answered
                if (state.selectedAnswers.every(a => a !== null)) {
                    showResults();
                } else {
                    alert('Please answer all questions before viewing results.');
                }
                return;
            }

            if (state.selectedAnswers[idx] === null) {
                alert('Please select an answer first.');
                return;
            }

            state.currentIndex = idx + 1;
            renderQuestion();
        }

        function prevQuestion() {
            if (state.currentIndex > 0) {
                state.currentIndex--;
                renderQuestion();
            }
        }

        function showResults() {
            const qs = state.questions;
            let correct = 0;
            qs.forEach((q, i) => {
                if (state.selectedAnswers[i] === q.ans) correct++;
            });
            state.score = correct;
            const total = qs.length;
            const pct = Math.round((correct / total) * 100);

            $('resultScore').textContent = pct;
            $('resultCorrect').textContent = correct;
            $('resultWrong').textContent = total - correct;
            $('resultTotal').textContent = total;

            let msg = 'Great job! 🎉';
            if (pct < 40) msg = 'Keep practicing! 💪';
            else if (pct < 70) msg = 'Good effort! 🌟';
            else if (pct < 90) msg = 'Well done! 👏';
            else msg = 'Excellent! 🏆';

            $('resultMessage').textContent = msg;

            showView('results');
        }

        function retryQuiz() {
            // reset and restart with same settings
            const level = state.level;
            const subject = state.subject;
            const term = state.term;
            const cls = state.class;

            state.questions = getQuestions(level, subject, term);
            state.currentIndex = 0;
            state.selectedAnswers = new Array(state.questions.length).fill(null);
            state.score = 0;
            state.totalQuestions = state.questions.length;

            showView('quiz');
            renderQuestion();
        }

        // ═══════════════════════════════════════════════════════════════════
        //  NOTES VIEW
        // ═══════════════════════════════════════════════════════════════════

        function showNotesView() {
            const levelLabel = state.level === 'primary' ? 'Primary' : state.class || 'Secondary';
            const subjectLabel = state.subject;
            const termLabel = state.term;

            $('notesTitle').textContent = `📖 ${subjectLabel} Notes`;
            $('notesSubjectTag').textContent = `${levelLabel} · ${termLabel}`;

            const notesHtml = getNotes(state.level, state.subject, state.term);
            $('notesContent').innerHTML = notesHtml;

            showView('notes');
        }

        // ═══════════════════════════════════════════════════════════════════
        //  KEYBOARD SHORTCUTS
        // ═══════════════════════════════════════════════════════════════════

        document.addEventListener('keydown', function(e) {
            if (e.key === 'Escape') {
                if (modal.classList.contains('open')) closeModal();
            }
            // number keys 1-4 for quiz options
            if (state.view === 'quiz') {
                const num = parseInt(e.key);
                if (num >= 1 && num <= 4) {
                    const idx = num - 1;
                    const container = $('optionsContainer');
                    const opts = container.querySelectorAll('.qopt:not(.disabled)');
                    if (opts[idx]) opts[idx].click();
                }
                if (e.key === 'Enter') {
                    const nextBtn = $('nextBtn');
                    if (!nextBtn.disabled) nextBtn.click();
                }
                if (e.key === 'ArrowLeft') $('prevBtn').click();
                if (e.key === 'ArrowRight') { if (!$('nextBtn').disabled) $('nextBtn').click(); }
            }
        });

        // ═══════════════════════════════════════════════════════════════════
        //  INIT
